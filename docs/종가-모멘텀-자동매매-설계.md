## 전략 핵심 원리

```
오늘 급등 신호로 강하게 마감한 종목
  → 다음 거래일 추가 상승 기대 (모멘텀 지속성)
  → 장 시작 5~15분 대기 후 방향성 확인 시 매수 (시초가 휩소 회피)
  → 목표가 도달 시 50% 분할 익절 & 잔여 물량 트레일링 스탑
```

**적중 조건 (이 전략이 유효한 경우)**
- 거래량 급증 + 종가 강세(양봉 + 고가 근처 종가)
- 박스권/52주 고가 돌파 당일
- 공시·계약 등 실제 재료 존재
- 기관/외국인 순매수 동반

**손실 조건 (이 전략이 실패하는 경우)**
- 당일 장대 윗꼬리 → 다음날 갭하락
- 단순 테마/풍문성 급등 → 세력 매도
- 시장 전체 급락일

## 전체 실행 흐름

```
[매일 15:50 KST] ──────────────────────────────
  1. pykrx로 당일 OHLCV 수집
  2. 급등 신호 스크리닝 실행 (기존 알고리즘)
  3. 후보 종목 Top 5 선정 및 저장

[매일 08:50 KST 다음 거래일] ───────────────────
  4. 후보 종목별 조건부 매수 감시 모드 돌입
  5. 종목당 투자금 = 총자산 / 종목 수 (A등급 주도주 비중 상향)

[장중 09:00 ~ 15:30] ───────────────────────────
  6. KIS WebSocket으로 실시간 체결가 감시
  7. 개장 직후 5~15분 방향성 및 거래대금 확인 후 매수 체결
  8. 목표가 도달 시 50% 즉시 익절 후 트레일링 스탑 모드
  9. 손절가 도달 → 전량 즉시 매도
  10. 15:20 강제 청산 (오버나잇 리스크 제거)
```

## PHASE 1. 종가 스크리닝 (15:50 실행)

```python
# screener.py
import pykrx.stock as stock
import pandas as pd
from datetime import date, timedelta

def get_eod_candidates(today: str, top_n: int = 5) -> pd.DataFrame:
    """
    장 마감 후 급등 신호 후보 종목 추출
    today: 'YYYYMMDD' 형식
    """
    # 1. 전체 종목 OHLCV 수집
    kospi = stock.get_market_ohlcv(today, market="KOSPI")
    kosdaq = stock.get_market_ohlcv(today, market="KOSDAQ")
    df = pd.concat([kospi, kosdaq])

    # 2. 기본 필터
    df = df[df['거래량'] > 0]
    df['등락률'] = df['등락률'].astype(float)

    # 3. 1차 컷: 당일 +3% 이상 상승 종목만
    df = df[df['등락률'] >= 3.0]

    # 4. 거래량 급증 필터 (20일 평균 대비 5배 이상)
    candidates = []
    for code in df.index:
        try:
            hist = stock.get_market_ohlcv_by_date(
                (date.today() - timedelta(days=30)).strftime('%Y%m%d'),
                today, code
            )
            if len(hist) < 5:
                continue

            avg_vol_20d = hist['거래량'].iloc[:-1].tail(20).mean()
            today_vol   = hist['거래량'].iloc[-1]
            vol_ratio   = today_vol / avg_vol_20d if avg_vol_20d > 0 else 0

            # 종가 강도: 고가 대비 종가 위치 (0~1, 높을수록 강함)
            row = hist.iloc[-1]
            price_range = row['고가'] - row['저가']
            close_strength = (row['종가'] - row['저가']) / price_range if price_range > 0 else 0.5

            # 윗꼬리 비율 (양봉 기준)
            body = row['종가'] - row['시가']
            upper_wick = row['고가'] - row['종가']
            wick_ratio = upper_wick / body if body > 0 else 99

            # 조건: 거래량 급증 + 종가 강도 높음 + 윗꼬리 과다 아님
            if vol_ratio >= 5.0 and close_strength >= 0.7 and wick_ratio < 1.5:
                candidates.append({
                    'code':            code,
                    'close':           row['종가'],
                    'change_pct':      df.loc[code, '등락률'],
                    'vol_ratio':       round(vol_ratio, 1),
                    'close_strength':  round(close_strength, 2),
                    'wick_ratio':      round(wick_ratio, 2),
                    # 매수 기준가: 당일 종가 기준
                    'entry_price':     row['종가'],
                    # 목표가: +4% (3거래일 내 목표)
                    'target_price':    round(row['종가'] * 1.04),
                    # 손절가: -2.5%
                    'stop_loss_price': round(row['종가'] * 0.975),
                })
        except Exception:
            continue

    result = pd.DataFrame(candidates)
    if result.empty:
        return result

    # 종가 강도 + 거래량 비율 기준 정렬 후 상위 N개
    result['score'] = result['close_strength'] * 0.6 + (result['vol_ratio'].clip(0, 20) / 20) * 0.4
    result = result.sort_values('score', ascending=False).head(top_n)

    return result.reset_index(drop=True)
```

## PHASE 2. 다음날 매수 주문 (08:50 실행)

```python
# trader.py
from kis_api import KISTrader  # 한국투자증권 API 래퍼

def place_buy_orders(candidates: pd.DataFrame, total_budget: int):
    """
    후보 종목 균등 분배 매수 주문
    total_budget: 총 투자 가능 금액 (원)
    """
    trader = KISTrader()
    n = len(candidates)
    budget_per_stock = total_budget // n

    orders = []
    for _, row in candidates.iterrows():
        qty = budget_per_stock // row['entry_price']
        if qty < 1:
            continue

        # 동시호가 매수 (시장가)
        result = trader.buy_market_order(
            code=row['code'],
            qty=qty,
        )
        orders.append({
            'code':             row['code'],
            'qty':              qty,
            'entry_price':      row['entry_price'],
            'target_price':     row['target_price'],
            'stop_loss_price':  row['stop_loss_price'],
            'order_id':         result['order_id'],
        })
        print(f"✅ 매수 주문: {row['code']} {qty}주 @ 시장가")

    return orders
```

## PHASE 3. 장중 실시간 감시 및 자동 매도 (09:00~15:20)

```python
# monitor.py
import asyncio
from datetime import datetime
from kis_api import KISTrader, KISWebSocket

class PositionMonitor:
    """
    보유 종목 실시간 감시 + 조건부 자동 매도
    """
    def __init__(self, positions: list[dict]):
        self.positions = {p['code']: p for p in positions}
        self.trader = KISTrader()
        self.sold = set()

    async def on_price_update(self, code: str, current_price: int):
        """WebSocket 체결가 수신 시 호출"""
        if code not in self.positions or code in self.sold:
            return

        pos = self.positions[code]
        target    = pos['target_price']
        stop_loss = pos['stop_loss_price']
        entry     = pos['entry_price']
        qty       = pos['qty']
        pnl_pct   = (current_price - entry) / entry * 100

        # 목표가 도달 → 익절
        if current_price >= target:
            await self._sell(code, qty, reason=f"목표가 도달 (+{pnl_pct:.1f}%)")

        # 손절가 도달 → 손절
        elif current_price <= stop_loss:
            await self._sell(code, qty, reason=f"손절가 도달 ({pnl_pct:.1f}%)")

    async def force_close_all(self):
        """15:20 강제 전량 청산 (오버나잇 방지)"""
        for code, pos in self.positions.items():
            if code not in self.sold:
                await self._sell(code, pos['qty'], reason="장마감 강제 청산")

    async def _sell(self, code: str, qty: int, reason: str):
        self.sold.add(code)
        self.trader.sell_market_order(code=code, qty=qty)
        print(f"🔴 매도: {code} {qty}주 | 사유: {reason}")

    async def run(self):
        ws = KISWebSocket()
        await ws.subscribe(list(self.positions.keys()), self.on_price_update)

        # 15:20에 강제 청산 트리거
        while datetime.now().strftime('%H:%M') < '15:20':
            await asyncio.sleep(1)

        await self.force_close_all()
```

## PHASE 4. 전체 스케줄러 통합

```python
# main.py
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from screener import get_eod_candidates
from trader import place_buy_orders
from monitor import PositionMonitor
import asyncio, json

TOTAL_BUDGET = 5_000_000  # 500만 원
CANDIDATE_FILE = 'data/candidates_today.json'

scheduler = AsyncIOScheduler(timezone='Asia/Seoul')
active_monitor: PositionMonitor | None = None

@scheduler.scheduled_job('cron', day_of_week='mon-fri', hour=15, minute=50)
def step1_screening():
    """15:50 스크리닝"""
    today = datetime.now().strftime('%Y%m%d')
    candidates = get_eod_candidates(today, top_n=5)

    if candidates.empty:
        print("⚠️ 오늘 급등 후보 없음")
        return

    candidates.to_json(CANDIDATE_FILE, force_ascii=False, orient='records')
    print(f"📌 후보 {len(candidates)}종목 저장 완료")
    print(candidates[['code', 'change_pct', 'vol_ratio', 'target_price', 'stop_loss_price']])

@scheduler.scheduled_job('cron', day_of_week='mon-fri', hour=8, minute=50)
async def step2_buy():
    """08:50 매수 주문"""
    global active_monitor

    try:
        candidates = pd.read_json(CANDIDATE_FILE, orient='records')
    except FileNotFoundError:
        print("⚠️ 후보 파일 없음, 매수 건너뜀")
        return

    positions = place_buy_orders(candidates, TOTAL_BUDGET)
    active_monitor = PositionMonitor(positions)

    print(f"🚀 {len(positions)}종목 매수 완료, 감시 시작")
    await active_monitor.run()

scheduler.start()
asyncio.get_event_loop().run_forever()
```

## 전략 성과 기대치 (백테스트 기준)

| 지표 | 예상치 | 비고 |
|---|---|---|
| 승률 | 50~60% | 강한 재료 필터 적용 시 |
| 평균 수익 (익절) | +3.5~4.5% | 목표가 기준 |
| 평균 손실 (손절) | -2~2.5% | 손절 엄수 시 |
| 기대값 | +0.5~1.5% / 거래 | 승률 55% 가정 |
| 최대 연속 손실 | 3~5회 | 드로우다운 구간 |

> **핵심**: 손절 엄수 여부가 전략 생존의 핵심. 손절 안 하면 전략 붕괴.
