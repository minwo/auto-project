## 전체 파이프라인 흐름

```
[STEP 1] 유니버스 필터링       → 종목 풀 구성
[STEP 2] 거래량 신호 평가      → volume_score (0~30점)
[STEP 3] 가격 패턴 신호 평가   → price_score  (0~25점)
[STEP 4] 기술적 지표 평가      → tech_score   (0~20점)
[STEP 5] 수급 신호 평가        → supply_score (0~15점)
[STEP 6] 공시/이벤트 평가      → event_score  (0~10점)
[STEP 7] 총점 산출 + 패널티    → total_score  (0~100점)
[STEP 8] Top N 후보 선별       → 결과 저장 / 알림 발송
```

## STEP 1. 유니버스 필터링

```python
def build_universe(df: pd.DataFrame) -> pd.DataFrame:
    """
    급등 감지 대상 종목 풀 구성
    입력: 전체 상장 종목 DataFrame
    출력: 조건을 통과한 종목 DataFrame
    """
    universe = df.copy()

    # 보통주만 포함 (ETF, ETN, 우선주, 스팩 제외)
    universe = universe[universe['stock_type'] == 'common']

    # 시가총액 1,000억 원 이상
    universe = universe[universe['market_cap'] >= 100_000_000_000]

    # 관리종목 / 투자유의·경고·위험 / 거래정지 제외
    universe = universe[universe['market_alert'].isin(['normal'])]
    universe = universe[universe['trading_halt'] == False]

    # 상장 60거래일 미만 제외
    universe = universe[universe['listed_days'] >= 60]

    # 최근 20거래일 평균 거래대금 10억 미만 제외 (저유동성)
    universe = universe[universe['avg_trading_value_20d'] >= 1_000_000_000]

    # 최근 6개월 내 유상증자 이력 종목 플래그 (제외 또는 주의 표시)
    universe['rights_offering_warning'] = universe['last_rights_offering_days'] < 120

    return universe
```

## STEP 2. 거래량 신호 평가 (최대 30점)

```python
def calc_volume_score(stock: dict) -> tuple[int, list[str]]:
    """
    거래량 관련 신호 점수 산출
    반환: (점수, 감지된 신호 리스트)
    """
    score = 0
    signals = []

    vol_today    = stock['volume_today']          # 당일 거래량
    vol_prev     = stock['volume_prev']           # 전일 거래량
    vol_avg_20d  = stock['volume_avg_20d']        # 20일 평균 거래량
    val_today    = stock['trading_value_today']   # 당일 거래대금
    val_avg_20d  = stock['trading_value_avg_20d'] # 20일 평균 거래대금
    short_ratio  = stock['short_ratio_today']     # 당일 공매도 비율
    short_ratio_prev_week = stock['short_ratio_prev_week']

    # 1) 전일 대비 거래량 급증 (최대 10점)
    vol_ratio_prev = vol_today / vol_prev if vol_prev > 0 else 0
    if vol_ratio_prev >= 10:
        score += 10; signals.append("거래량 전일 대비 1000% 이상 급증")
    elif vol_ratio_prev >= 5:
        score += 7;  signals.append("거래량 전일 대비 500% 이상 급증")
    elif vol_ratio_prev >= 3:
        score += 5;  signals.append("거래량 전일 대비 300% 이상 급증")

    # 2) 20일 평균 대비 거래대금 돌파 (최대 10점)
    val_ratio_20d = val_today / val_avg_20d if val_avg_20d > 0 else 0
    if val_ratio_20d >= 10:
        score += 10; signals.append("거래대금 20일 평균 대비 1000% 초과")
    elif val_ratio_20d >= 5:
        score += 7;  signals.append("거래대금 20일 평균 대비 500% 초과")
    elif val_ratio_20d >= 3:
        score += 4;  signals.append("거래대금 20일 평균 대비 300% 초과")

    # 3) 20일 평균 거래량 대비 당일 거래량 (최대 5점)
    vol_ratio_20d = vol_today / vol_avg_20d if vol_avg_20d > 0 else 0
    if vol_ratio_20d >= 5:
        score += 5; signals.append("거래량 20일 평균 대비 500% 초과")
    elif vol_ratio_20d >= 3:
        score += 3; signals.append("거래량 20일 평균 대비 300% 초과")

    # 4) 공매도 감소 + 거래량 증가 (최대 5점)
    if short_ratio_prev_week > 0:
        short_decrease = (short_ratio_prev_week - short_ratio) / short_ratio_prev_week
        if short_decrease >= 0.3 and vol_ratio_20d >= 3:
            score += 5; signals.append("공매도 30% 감소 + 거래량 급증 (숏커버링)")

    return min(score, 30), signals
```

## STEP 3. 가격 패턴 신호 평가 (최대 25점)

```python
def calc_price_score(stock: dict, ohlcv: pd.DataFrame) -> tuple[int, list[str]]:
    """
    가격 패턴 신호 점수 산출
    ohlcv: 최근 N일 OHLCV DataFrame (index: date, columns: open/high/low/close/volume)
    """
    score = 0
    signals = []

    close_today  = stock['close_today']
    open_today   = stock['open_today']
    close_prev   = stock['close_prev']
    high_52w     = stock['high_52w']       # 52주 최고가
    low_52w      = stock['low_52w']        # 52주 최저가

    # 1) 52주 신고가 갱신 (7점)
    if close_today >= high_52w:
        score += 7; signals.append("52주 신고가 갱신")

    # 2) 52주 최고가 대비 80% 이상 회복 (3점)
    elif close_today >= high_52w * 0.8:
        score += 3; signals.append("52주 고가 대비 80% 이상 회복")

    # 3) 박스권 돌파 (6점)
    #    최근 20일 종가의 최고점을 당일 종가가 돌파하는 경우
    box_top = ohlcv['close'].iloc[-21:-1].max()  # 직전 20일 최고 종가
    if close_today > box_top and (close_today - box_top) / box_top >= 0.01:
        score += 6; signals.append(f"박스권 상단({box_top:,.0f}원) 돌파")

    # 4) 갭 상승 시가 (4점)
    gap_ratio = (open_today - close_prev) / close_prev
    if gap_ratio >= 0.05:
        score += 4; signals.append(f"갭 상승 시가 ({gap_ratio*100:.1f}%)")
    elif gap_ratio >= 0.03:
        score += 2; signals.append(f"갭 상승 시가 ({gap_ratio*100:.1f}%)")

    # 5) 연속 양봉 (최대 5점)
    #    최근 N일 연속 양봉 (당일 포함)
    recent = ohlcv.tail(5)
    consec_bull = 0
    for _, row in recent.iterrows():
        if row['close'] > row['open']:
            consec_bull += 1
        else:
            consec_bull = 0
    if consec_bull >= 5:
        score += 5; signals.append("5일 연속 양봉")
    elif consec_bull >= 3:
        score += 3; signals.append("3일 연속 양봉")

    # 6) 눌림목 반등: 5일선 또는 20일선 지지 후 당일 반등
    ma5  = ohlcv['close'].tail(5).mean()
    ma20 = ohlcv['close'].tail(20).mean()
    low_today = stock['low_today']
    if low_today <= ma5 * 1.01 and close_today > ma5:
        score += 3; signals.append("5일선 지지 후 반등")
    elif low_today <= ma20 * 1.01 and close_today > ma20:
        score += 2; signals.append("20일선 지지 후 반등")

    return min(score, 25), signals
```

import pandas_ta as ta

def calc_tech_score(ohlcv: pd.DataFrame) -> tuple[int, list[str]]:
    """
    기술적 지표 신호 점수 산출
    ohlcv: 최근 120일 이상 OHLCV DataFrame
    """
    score = 0
    signals = []

    close = ohlcv['close']
    high  = ohlcv['high']
    low   = ohlcv['low']
    volume = ohlcv['volume']

    # --- RSI (최대 5점) ---
    rsi = ta.rsi(close, length=14)
    rsi_today = rsi.iloc[-1]
    rsi_prev  = rsi.iloc[-2]

    if 30 <= rsi_prev < 35 and rsi_today >= 35:   # 과매도 탈출
        score += 5; signals.append(f"RSI 과매도 탈출 ({rsi_today:.1f})")
    elif 45 <= rsi_prev < 50 and rsi_today >= 50:  # 50선 돌파
        score += 3; signals.append(f"RSI 50선 돌파 ({rsi_today:.1f})")
    elif 40 <= rsi_today <= 70:                    # 건전한 상승 구간
        score += 1

    # --- MACD 골든크로스 (최대 5점) ---
    macd_df = ta.macd(close, fast=12, slow=26, signal=9)
    macd_line   = macd_df['MACD_12_26_9']
    signal_line = macd_df['MACDs_12_26_9']
    histogram   = macd_df['MACDh_12_26_9']

    if macd_line.iloc[-2] < signal_line.iloc[-2] and macd_line.iloc[-1] >= signal_line.iloc[-1]:
        score += 4; signals.append("MACD 골든크로스")
    if histogram.iloc[-1] > histogram.iloc[-2] > 0:
        score += 1; signals.append("MACD 히스토그램 확대")

    # --- 볼린저밴드 돌파 (최대 4점) ---
    bb = ta.bbands(close, length=20, std=2)
    bb_upper = bb['BBU_20_2.0']
    bb_width_today = (bb['BBU_20_2.0'] - bb['BBL_20_2.0']).iloc[-1]
    bb_width_prev  = (bb['BBU_20_2.0'] - bb['BBL_20_2.0']).iloc[-6:-1].mean()
    bb_pct = bb['BBP_20_2.0'].iloc[-1]   # %B 지표

    squeeze = bb_width_today < bb_width_prev * 0.7  # 밴드 수축 여부
    if bb_pct >= 1.0:
        score += 4 if squeeze else 2
        signals.append("볼린저밴드 상단 돌파" + (" (스퀴즈 후)" if squeeze else ""))

    # --- 이동평균선 정배열 (최대 4점) ---
    ma5  = close.tail(5).mean()
    ma20 = close.tail(20).mean()
    ma60 = close.tail(60).mean()

    if ma5 > ma20 > ma60:
        score += 4; signals.append("5/20/60일선 정배열")
    elif ma5 > ma20:
        score += 2; signals.append("5/20일선 정배열")

    # --- OBV 상승 추세 (최대 2점) ---
    obv = ta.obv(close, volume)
    obv_trend = obv.iloc[-5:].is_monotonic_increasing  # 최근 5일 OBV 상승
    price_flat = (close.iloc[-5:].max() - close.iloc[-5:].min()) / close.iloc[-5].mean() < 0.03

    if obv_trend and price_flat:
        score += 2; signals.append("OBV 지속 상승 (횡보 중 매집 신호)")

    return min(score, 20), signals
```

def calc_supply_score(stock: dict) -> tuple[int, list[str]]:
    """
    수급/세력 신호 점수 산출
    """
    score = 0
    signals = []

    inst_net_buy_3d  = stock['inst_net_buy_3d']    # 최근 3거래일 기관 순매수 (원)
    foreign_net_buy_3d = stock['foreign_net_buy_3d']# 최근 3거래일 외국인 순매수 (원)
    program_buy_today = stock['program_buy_net']    # 당일 프로그램 순매수
    short_balance_chg = stock['short_balance_chg_pct']  # 대차잔고 변화율

    # 1) 기관 순매수 전환 (최대 6점)
    if inst_net_buy_3d > 0:
        score += 6 if inst_net_buy_3d > 5_000_000_000 else 3
        signals.append(f"기관 순매수 전환 ({inst_net_buy_3d/1e8:.0f}억)")

    # 2) 외국인 순매수 전환 (최대 5점)
    if foreign_net_buy_3d > 0:
        score += 5 if foreign_net_buy_3d > 3_000_000_000 else 2
        signals.append(f"외국인 순매수 전환 ({foreign_net_buy_3d/1e8:.0f}억)")

    # 3) 프로그램 순매수 (2점)
    if program_buy_today > 0:
        score += 2; signals.append("프로그램 순매수")

    # 4) 대차잔고 감소 (2점)
    if short_balance_chg <= -0.10:
        score += 2; signals.append(f"대차잔고 감소 ({short_balance_chg*100:.1f}%)")

    return min(score, 15), signals
```

## STEP 6. 공시/이벤트 신호 평가 (최대 10점)

```python
def calc_event_score(disclosures: list[dict]) -> tuple[int, list[str]]:
    """
    당일 DART 공시 기반 이벤트 신호 점수 산출
    disclosures: DART OpenAPI에서 수집한 당일 공시 목록
    """
    score = 0
    signals = []

    # 공시 키워드 → 점수 매핑
    EVENT_SCORE_MAP = {
        # 키워드: (점수, 신호 설명)
        "매출액또는손익구조30%이상변경":  (10, "실적 서프라이즈"),
        "단일판매·공급계약체결":          (10, "대형 계약 체결"),
        "주요사항보고서":                 (6,  "주요사항 공시"),
        "최대주주변경":                   (8,  "최대주주 변경"),
        "임원·주요주주특정증권등소유상황보고서": (5, "임원 자사주 매입"),
        "유상증자결정철회":               (6,  "유상증자 철회"),
        "특허권취득":                     (5,  "특허 등록"),
        "기술이전":                       (5,  "기술 이전 계약"),
        "지수편입":                       (7,  "지수 편입 예고"),
    }

    for disc in disclosures:
        title = disc.get('report_nm', '')
        for keyword, (pts, label) in EVENT_SCORE_MAP.items():
            if keyword in title.replace(' ', ''):
                score += pts
                signals.append(f"공시: {label}")
                break  # 동일 공시에 중복 합산 방지

    return min(score, 10), signals
```

## STEP 7. 총점 산출 + 패널티 적용

```python
def calc_total_score(
    volume_score:  int,
    price_score:   int,
    tech_score:    int,
    supply_score:  int,
    event_score:   int,
    stock:         dict,
) -> tuple[int, list[str]]:
    """
    종합 점수 산출 및 리스크 패널티 적용
    반환: (최종 점수, 리스크 플래그 리스트)
    """
    raw_score = volume_score + price_score + tech_score + supply_score + event_score
    penalty   = 0
    risk_flags = []

    # --- 리스크 패널티 항목 ---

    # 1) 투자경고/위험 종목
    if stock['market_alert'] in ['warning', 'danger']:
        penalty += 20; risk_flags.append("투자경고/위험 종목")

    # 2) 장대 윗꼬리 (세력 매도 가능성)
    high  = stock['high_today']
    close = stock['close_today']
    open_ = stock['open_today']
    body  = abs(close - open_)
    upper_wick = high - max(close, open_)
    if body > 0 and upper_wick / body > 2.0:
        penalty += 10; risk_flags.append("장대 윗꼬리 (매도세 강함)")

    # 3) 과도한 갭 상승 (갭 10% 초과)
    gap_ratio = (stock['open_today'] - stock['close_prev']) / stock['close_prev']
    if gap_ratio >= 0.10:
        penalty += 8; risk_flags.append(f"과도한 갭상승 ({gap_ratio*100:.1f}%)")

    # 4) RSI 과매수 (70 초과)
    if stock.get('rsi_today', 50) > 70:
        penalty += 5; risk_flags.append(f"RSI 과매수 ({stock['rsi_today']:.1f})")

    # 5) 유상증자 이력 주의
    if stock.get('rights_offering_warning'):
        penalty += 5; risk_flags.append("최근 6개월 내 유상증자 이력")

    total = max(0, raw_score - penalty)
    return total, risk_flags
```

## STEP 8. 전체 스크리닝 실행 + Top N 선별

```python
def run_screening(
    universe:     pd.DataFrame,
    ohlcv_map:    dict,   # {code: pd.DataFrame}
    supply_map:   dict,   # {code: dict}
    disc_map:     dict,   # {code: list[dict]}
    top_n:        int = 10,
    min_score:    int = 60,
    max_per_sector: int = 3,
) -> pd.DataFrame:
    """
    전체 급등 신호 스크리닝 실행
    반환: 상위 후보 DataFrame
    """
    results = []

    for _, row in universe.iterrows():
        code  = row['code']
        stock = row.to_dict()
        ohlcv = ohlcv_map.get(code)

        if ohlcv is None or len(ohlcv) < 60:
            continue

        # --- 기본 필터 (1차 컷) ---
        price_chg = (stock['close_today'] - stock['close_prev']) / stock['close_prev']
        vol_ratio = stock['volume_today'] / stock['volume_avg_20d']
        if price_chg < 0.03 or vol_ratio < 5.0:
            continue

        # --- 신호별 점수 산출 ---
        v_score, v_signals = calc_volume_score(stock)
        p_score, p_signals = calc_price_score(stock, ohlcv)
        t_score, t_signals = calc_tech_score(ohlcv)
        s_score, s_signals = calc_supply_score(supply_map.get(code, {}))
        e_score, e_signals = calc_event_score(disc_map.get(code, []))

        total, risk_flags = calc_total_score(v_score, p_score, t_score, s_score, e_score, stock)

        all_signals = v_signals + p_signals + t_signals + s_signals + e_signals

        results.append({
            'code':          code,
            'name':          stock['name'],
            'sector':        stock['sector'],
            'total_score':   total,
            'volume_score':  v_score,
            'price_score':   p_score,
            'tech_score':    t_score,
            'supply_score':  s_score,
            'event_score':   e_score,
            'signals':       all_signals[:3],    # 상위 3개 이유
            'risk_flags':    risk_flags[:2],      # 상위 2개 위험 신호
            'price_change':  price_chg,
            'vol_ratio':     vol_ratio,
        })

    if not results:
        return pd.DataFrame()

    df = pd.DataFrame(results)

    # 최소 점수 필터
    df = df[df['total_score'] >= min_score]

    # 점수 내림차순 정렬
    df = df.sort_values('total_score', ascending=False)

    # 섹터 쏠림 방지: 동일 섹터 최대 3종목
    final = []
    sector_count: dict[str, int] = {}
    for _, row in df.iterrows():
        sector = row['sector']
        if sector_count.get(sector, 0) < max_per_sector:
            final.append(row)
            sector_count[sector] = sector_count.get(sector, 0) + 1
        if len(final) >= top_n:
            break

    return pd.DataFrame(final).reset_index(drop=True)
```

scheduler.start()
